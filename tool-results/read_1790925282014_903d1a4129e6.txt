   293→const $ = (sel) => document.querySelector(sel);

   295→const el = (tag, props = {}, ...children) => {
   296→  const n = document.createElement(tag);
   297→  Object.entries(props).forEach(([k, v]) => {
   298→    if (k === 'class') n.className = v;
   299→    else if (k === 'html') n.innerHTML = v;
   300→    else if (k === 'text') n.textContent = v;
   301→    else if (k.startsWith('on') && typeof v === 'function') n.addEventListener(k.slice(2).toLowerCase(), v);
   302→    else if (k === 'dataset') Object.entries(v).forEach(([dk, dv]) => n.dataset[dk] = dv);
   303→    else if (v !== null && v !== undefined) n.setAttribute(k, v);
   304→  });
   305→  children.forEach((c) => {
   306→    if (c === null || c === undefined) return;
   307→    if (typeof c === 'string' || typeof c === 'number') n.appendChild(document.createTextNode(String(c)));
   308→    else n.appendChild(c);
   309→  });
   310→  return n;

   325→function loadCart() {
   326→  try { return JSON.parse(localStorage.getItem(CART_KEY)) || []; }
   327→  catch { return []; }
   328→}
   329→function saveCart() { localStorage.setItem(CART_KEY, JSON.stringify(cart)); }
   330→
   331→function cartTotalItems() { return cart.reduce((s, i) => s + i.quantity, 0); }
   332→function cartTotalPrice() { return cart.reduce((s, i) => s + i.product.price * i.quantity, 0); }
   333→
   334→function addToCart(product, color, qty = 1) {
   335→  const idx = cart.findIndex(i => i.product.id === product.id && i.color === color);
   336→  if (idx >= 0) cart[idx].quantity += qty;
   337→  else cart.push({ product, color, quantity: qty });
   338→  saveCart();
   339→  renderCartCount();
   340→  openCart();
   341→  showToast(`Added ${product.name} (${color})`);
   342→}
   343→
   344→function removeFromCart(productId, color) {
   345→  cart = cart.filter(i => !(i.product.id === productId && i.color === color));
   346→  saveCart();
   347→  renderCartCount();
   348→  renderCartItems();
   349→}
   350→
   351→function updateQty(productId, color, qty) {
   352→  const item = cart.find(i => i.product.id === productId && i.color === color);
   353→  if (!item) return;
   354→  item.quantity = Math.max(1, qty);
   355→  if (item.quantity <= 0) removeFromCart(productId, color);
   356→  saveCart();
   357→  renderCartCount();
   358→  renderCartItems();
   359→}

   701→/* ============================================================ Scroll Reveal ============================================================ */
   702→let revealObserver;
   703→function observeReveals() {
   704→  if (!revealObserver) {
   705→    revealObserver = new IntersectionObserver((entries) => {
   706→      entries.forEach((e) => {
   707→        if (e.isIntersecting) {
   708→          e.target.classList.add('revealed');
   709→          revealObserver.unobserve(e.target);
   710→        }
   711→      });
   712→    }, { threshold: 0.12, rootMargin: '0px 0px -80px 0px' });
   713→  }
   714→  $$('.reveal:not(.revealed)').forEach((node) => revealObserver.observe(node));
   715→}
